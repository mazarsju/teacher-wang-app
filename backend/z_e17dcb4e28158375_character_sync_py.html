52def extract_tone_syllables_in_order(text: str, count: int) -> list[str] | None: 

53 """Scan ``text`` for up to ``count`` pinyin syllables, in order. 

54 

55 Mirrors the frontend's ``extractToneSyllablesInOrder``: a syllable is 

56 lowercase (tone digit optional -- neutral-tone particles like "le" and 

57 "ma" are common); anything else -- letters, digits, punctuation, 

58 whitespace -- is skipped. Used to pull each Han character's pinyin out 

59 of a word's pinyin field when that field also holds non-Han filler 

60 (parentheses, ellipses, question marks, ...) that doesn't need to line 

61 up positionally with real characters. 

62 

63 ponytail: requiring the scan to start on a lowercase letter keeps a 

64 stray placeholder like "A" from being mistaken for the bare pinyin 

65 final "a"; a pinyin-shaped substring hiding inside longer filler can 

66 still false-match, which is acceptable since the filler itself is 

67 intentionally unchecked. 

68 

69 Candidates are validated through ``normalize_anki_pinyin_token`` (not 

70 ``is_valid_pinyin`` directly), since real Anki pinyin fields are almost 

71 always accented ("gōng", "cóng") and ``is_valid_pinyin`` only accepts 

72 already-normalized ASCII/tone-digit syllables -- checking accented text 

73 against it directly would reject every real syllable. The normalized 

74 (tone-digit) form is what gets stored, matching the other resolution 

75 path in ``_resolved_word_tokens``. 

76 """ 

77 matches: list[str] = [] 

78 i = 0 

79 n = len(text) 

80 while i < n and len(matches) < count: 

81 if text[i] not in _PINYIN_START_CHARS: 

82 i += 1 

83 continue 

84 matched_length = 0 

85 matched_token: str | None = None 

86 for length in range(min(PINYIN_MAX_LENGTH, n - i), 0, -1): 86 ↛ 99line 86 didn't jump to line 99 because the loop on line 86 didn't complete

87 candidate = text[i : i + length] 

88 # Reject a candidate that only validates because normalization 

89 # trims it internally (e.g. "hao3 " -> "hao3"); the exact span 

90 # must already be clean so a trailing space on a longer 

91 # candidate can't shadow a clean, shorter match. 

92 if candidate[-1].isspace(): 

93 continue 

94 normalized = normalize_anki_pinyin_token(candidate) 

95 if normalized is not None: 

96 matched_length = length 

97 matched_token = normalized 

98 break

100 matches.append(matched_token) # type: ignore[arg-type] 

101 i += matched_length 

102 else: 

103 i += 1 

104 return matches if len(matches) == count else None

134 `AddWordModal`'s fallback). ``None`` marks a position that couldn't be 

135 resolved. When ``pinyin_field`` is blank there are no field tokens to 

136 align against, so non-Han characters pass through as their own literal. 

137 

138 A word mixed with non-Han characters (parentheses, ellipses, question

140 Anki fields routinely glue that punctuation directly onto the

142 index-aligned/literal-match rule below since it expects the field's

228def build_character_pinyin_map_from_words(words: list[Word]) -> dict[str, list[str]]: 

229 """Derive unique pinyin readings per Han character from word rows. 

230 

231 Words are processed in ``(updated_at, word)`` order. Within each word, 

232 characters are scanned left to right; the first occurrence of each Han 

233 character picks its syllable. A word made entirely of Han characters 

234 uses the syllable at the same index in the space-separated pinyin field 

235 (matching the frontend ``isWordPinyinValid`` rules); a word mixed with 

236 non-Han characters instead resolves each Han character's syllable via 

237 ``extract_tone_syllables_in_order``, since the field's spacing and 

238 non-Han content aren't required to line up positionally there. 

239 """ 

240 char_readings: dict[str, list[str]] = {} 

241 ordered_words = sorted(words, key=lambda row: (row.updated_at, row.word)) 

242 

243 for word in ordered_words: 

244 pinyin_field = word.pinyin or "" 

245 han_chars = [char for char in word.word if is_han_character(char)] 

246 has_non_han = len(han_chars) < len(word.word) 

247 

248 if has_non_han: 

249 han_readings = extract_tone_syllables_in_order( 

250 pinyin_field, len(han_chars) 

251 ) 

252 else: 

253 tokens = pinyin_field.split() 

254 han_readings = [ 

255 _valid_reading_at(tokens, index) for index in range(len(han_chars)) 

256 ] 

257 

258 seen_in_word: set[str] = set() 

259 han_index = 0 

260 

261 for char in word.word: 

262 if not is_han_character(char): 

263 continue 

264 reading = han_readings[han_index] if han_readings is not None else None 

265 han_index += 1 

266 if char in seen_in_word: 

267 continue 

268 seen_in_word.add(char) 

269 char_readings.setdefault(char, []) 

270 

271 if reading is not None and reading not in char_readings[char]: 

272 char_readings[char].append(reading) 

273 

274 return char_readings

288def rebuild_characters_from_words(user_id) -> CharacterSyncResult: 

289 """Synchronize ``character`` rows with all ``words`` for ``user_id``. 

290 

291 ``words`` is the source of truth: both ``pinyin_readings`` and 

292 ``writing_known`` are fully derived here, in either direction (a 

293 character reverts to not-writing-known once no remaining word marks it 

294 known). 

295 

296 Returns the characters that were created or modified (i.e. whose 

297 ``updated_at`` changed) and the ids of any characters deleted. 

298 """ 

299 words = Word.query.filter_by(user_id=user_id).all() 

300 target = build_character_pinyin_map_from_words(words) 

301 writing_known_map = build_character_writing_known_map_from_words(words) 

302 

303 existing = { 

304 row.char: row for row in Character.query.filter_by(user_id=user_id).all() 

305 } 

306 updated_characters: list[Character] = [] 

307 now = utcnow() 

308 

309 for char, readings in target.items(): 

310 record = existing.get(char) 

311 writing_known = writing_known_map.get(char, False) 

312 if record is None: 

313 record = Character( 

314 user_id=user_id, 

315 char=char, 

316 pinyin_readings=readings, 

317 writing_known=writing_known, 

318 updated_at=now, 

319 ) 

320 db.session.add(record) 

321 updated_characters.append(record) 

322 continue 

323 

324 changed = False 

325 if record.pinyin_readings != readings: 

326 record.pinyin_readings = readings 

327 changed = True 

328 

329 if record.writing_known != writing_known: 

330 record.writing_known = writing_known 

331 changed = True 

332 

333 if changed: 

334 updated_characters.append(record) 

335 

336 deleted_char_ids = [char for char in existing if char not in target] 

337 for char in deleted_char_ids: 

338 db.session.delete(existing[char]) 

339 

340 db.session.flush() 

341 return CharacterSyncResult( 

342 updated_characters=updated_characters, 

343 deleted_char_ids=deleted_char_ids, 

344 )
