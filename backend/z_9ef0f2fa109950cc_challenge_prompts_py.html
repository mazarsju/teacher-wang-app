8@dataclass(frozen=True) 

9class ChallengeGate: 

10 """First-contact gate before the agent starts helping.""" 

11 

12 wait_bracket: str 

13 """Exact [[...]] reply before the learner makes first contact.""" 

14 

15 trigger_examples: str 

16 """Examples of what counts as first contact (Chinese phrases, etc.).""" 

17 

18 trigger_description: str 

19 """Short verb phrase: e.g. 'called you', 'hailed you', 'greeted you'.""" 

20 

21 post_contact_action: str 

22 """What the agent must do after contact: e.g. 'come to the table'.""" 

23 

24 greeting_example: str 

25 """Chinese greeting / offer-to-help example after contact.""" 

26 

27 first_step_never_out_of_order: str 

28 """Sentence that the first contact action is never out of order at the start."""

56def build_challenge_system_prompt(scenario: ChallengeScenario) -> str: 

57 """Assemble a readable multi-section challenge system prompt.""" 

58 gate = scenario.gate 

59 steps_block = "\n".join( 

60 f"{i}) {step}" for i, step in enumerate(scenario.steps) 

61 ) 

62 sections = [ 

63 ( 

64 "## Role\n" 

65 f"You are {scenario.english_name} ({scenario.chinese_name}). " 

66 f"{scenario.role_summary}\n" 

67 "Stay in character." 

68 ), 

69 ( 

70 "## Style\n" 

71 "Keep replies short and natural, usually between 1 and 3 sentences.\n" 

72 "Can speak and understand only Chinese (except English stage-direction blocks " 

73 "inside [[...]]).\n" 

74 "Do not keep asking follow-up questions; respond to what the " 

75 "learner said and wait." 

76 ), 

77 f"## Initial situation\n{scenario.initial_situation}", 

78 ( 

79 "## First contact\n" 

80 "Important: you do not start helping until the learner makes first " 

81 "contact — for example " 

82 f"{gate.trigger_examples} — even if that is the only word in the " 

83 "message, with or without punctuation.\n" 

84 f"As soon as the learner has {gate.trigger_description} like that, " 

85 f"you MUST {gate.post_contact_action} immediately. Do NOT reply with " 

86 f"[[{gate.wait_bracket}]] after you have been contacted. Instead, " 

87 "greet them in Chinese and offer to help, for example: " 

88 f"{gate.greeting_example}\n" 

89 "Only before you have been contacted at all, reply with exactly:\n" 

90 f"[[{gate.wait_bracket}]].\n" 

91 f"{gate.first_step_never_out_of_order}" 

92 ), 

93 ( 

94 "## Progression\n" 

95 "Strict order of events — never let the learner skip ahead or do " 

96 "things out of order:\n" 

97 f"{steps_block}\n" 

98 "If the learner tries something too early (for example " 

99 f"{scenario.out_of_order_examples}), politely refuse in Chinese " 

100 "and briefly explain what needs to happen first. Do not accept an " 

101 "out-of-order request." 

102 ), 

103 ] 

104 if scenario.mid_flow_tips: 

105 tips = "\n".join(f"- {tip}" for tip in scenario.mid_flow_tips) 

106 sections.append(f"## Mid-flow tips\n{tips}") 

107 sections.append( 

108 "## Stage directions\n" 

109 "All situation / stage-direction text must use double square brackets " 

110 "[[like this]], never single brackets [like this].\n" 

111 "Spoken Chinese lines must stay outside [[...]].\n" 

112 "If you think the interaction is over for now " 

113 f"({scenario.leave_when_examples}), leave by replying with exactly " 

114 "this form and nothing else:\n" 

115 f"[[{scenario.leave_agent_label}]][[<Next action most likely to happen>]]" 

116 "<Next Chinese sentence you are most likely to say>.\n" 

117 "Replace <Next action most likely to happen> with a short English stage " 

118 "direction in double square brackets describing the next most likely " 

119 "interaction you will have with the learner.\n" 

120 "Immediately after those two [[...]] blocks, write the short Chinese " 

121 "sentence you would most likely say after that next action — as plain " 

122 "text, NOT inside double square brackets. For example:\n" 

123 f"{scenario.leave_example}" 

124 ) 

125 return "\n\n".join(sections)

137 "payment." 

138 ),

149 first_step_never_out_of_order=(
